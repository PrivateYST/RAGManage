- 权限控制（基于角色）

**技术实现**：
```python
# FastAPI + JWT
from fastapi import Depends, HTTPException
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import jwt, JWTError

security = HTTPBearer()

async def verify_token(credentials: HTTPAuthorizationCredentials = Depends(security)):
    token = credentials.credentials
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=["HS256"])
        return payload["user_id"]
    except JWTError:
        raise HTTPException(401, "无效的认证令牌")
```

**API Key 认证**：
```python
async def verify_api_key(x_api_key: str = Header(None)):
    if not x_api_key:
        raise HTTPException(401, "缺少 API Key")
    # 查询数据库验证 Key
    key_hash = hash_api_key(x_api_key)
    api_key = await db.get_api_key(key_hash)
    if not api_key or api_key.status != 1:
        raise HTTPException(401, "无效的 API Key")
    return api_key.user_id
```

### 2.2 文档管理模块

**核心流程**：
```
用户上传文档
     ↓
FastAPI 接收文件
     ↓
1. 保存到对象存储 (MinIO/S3)
2. 创建 documents 记录 (status=待处理)
3. 创建 Celery 任务
     ↓
返回任务 ID 给用户
     ↓
Celery Worker 异步处理：
  1. 根据文件类型解析内容
     - PDF: pdfplumber
     - Word: python-docx
     - Markdown: 直接读取
  2. 文本切分 (LangChain TextSplitter)
  3. 调用 Embedding 模型
  4. 写入 Milvus
  5. 更新 documents 记录 (status=完成)
     ↓
WebSocket 推送进度给前端
```

**文档切分策略**：
```python
from langchain.text_splitter import RecursiveCharacterTextSplitter

splitter = RecursiveCharacterTextSplitter(
    chunk_size=512,           # 每个块 512 字符
    chunk_overlap=50,         # 块之间重叠 50 字符
    separators=["\n\n", "\n", "。", "！", "？", " "]
)
```

**文档解析**：
```python
# PDF 解析
import pdfplumber
def parse_pdf(file_path):
    with pdfplumber.open(file_path) as pdf:
        text = "\n".join([page.extract_text() for page in pdf.pages])
    return text

# Word 解析
from docx import Document
def parse_docx(file_path):
    doc = Document(file_path)
    text = "\n".join([para.text for para in doc.paragraphs])
    return text
```

### 2.3 向量检索模块

**Milvus 多租户隔离**：
```python
# 每个用户一个 Collection
collection_name = f"user_{user_id}_vectors"

# 创建 Collection
from pymilvus import Collection, FieldSchema, CollectionSchema, DataType

fields = [
    FieldSchema(name="chunk_id", dtype=DataType.INT64, is_primary=True),
    FieldSchema(name="document_id", dtype=DataType.INT64),
    FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR, dim=1024),
    FieldSchema(name="content", dtype=DataType.VARCHAR, max_length=2000),
    FieldSchema(name="metadata", dtype=DataType.JSON),
]
schema = CollectionSchema(fields=fields)
collection = Collection(name=collection_name, schema=schema)

# 创建索引（HNSW）
index_params = {
    "metric_type": "IP",        # 内积（余弦相似度）
    "index_type": "HNSW",
    "params": {"M": 16, "efConstruction": 256}
}
collection.create_index(field_name="embedding", index_params=index_params)
```

**检索流程**：
```python
async def search_knowledge(user_id: int, query: str, top_k: int = 5):
    # 1. 查询向量化
    query_embedding = await embedding_service.embed(query)
    
    # 2. Milvus 检索
    collection_name = f"user_{user_id}_vectors"
    collection = Collection(collection_name)
    collection.load()
    
    results = collection.search(
        data=[query_embedding],
        anns_field="embedding",
        param={"metric_type": "IP", "params": {"ef": 64}},
        limit=top_k,
        output_fields=["chunk_id", "content", "metadata"]
    )
    
    # 3. 返回检索结果
    return [
        {
            "chunk_id": hit.entity.get("chunk_id"),
            "content": hit.entity.get("content"),
            "score": hit.distance,
            "metadata": hit.entity.get("metadata")
        }
        for hit in results[0]
    ]
```

### 2.4 对话生成模块

**RAG 流程**：
```python
async def chat(user_id: int, conversation_id: int, message: str):
    # 1. 检索相关文档
    retrieved_chunks = await search_knowledge(user_id, message, top_k=5)
    
    # 2. 构建 Prompt
    context = "\n\n".join([chunk["content"] for chunk in retrieved_chunks])
    prompt = f"""
基于以下知识库内容回答问题：

{context}

问题：{message}

请基于上述内容回答，如果内容中没有相关信息，请明确说明。
"""
    
    # 3. 调用 LLM（流式）
    async for chunk in llm_service.stream_chat(prompt):
        yield chunk  # 通过 WebSocket 实时返回
    
    # 4. 保存对话历史
    await db.save_message(conversation_id, "user", message)
    await db.save_message(conversation_id, "assistant", full_response, 
                         retrieved_chunks=retrieved_chunks)
```

**LLM 调用（vLLM）**：
```python
import httpx

async def stream_chat(prompt: str):
    async with httpx.AsyncClient() as client:
        async with client.stream(
            "POST",
            "http://localhost:8000/v1/completions",
            json={
                "model": "Qwen/Qwen2.5-7B-Instruct",
                "prompt": prompt,
                "max_tokens": 2000,
                "temperature": 0.7,
                "stream": True
            }
        ) as response:
            async for line in response.aiter_lines():
                if line.startswith("data: "):
                    data = json.loads(line[6:])
                    if "choices" in data:
                        yield data["choices"][0]["text"]
```

### 2.5 异步任务模块

**Celery 配置**：
```python
# celery_config.py
from celery import Celery

app = Celery(
    "rag_tasks",
    broker="redis://localhost:6379/0",
    backend="redis://localhost:6379/0"
)

app.conf.update(
    task_serializer="json",
    accept_content=["json"],
    result_serializer="json",
    timezone="UTC",
    enable_utc=True,
    task_track_started=True,
    task_time_limit=3600,  # 1小时超时
)
```

**文档处理任务**：
```python
from celery import Task
from celery.signals import task_prerun, task_postrun

class DocumentProcessTask(Task):
    def on_success(self, retval, task_id, args, kwargs):
        # 任务成功，通过 WebSocket 通知前端
        notify_user_via_websocket(user_id, {
            "task_id": task_id,
            "status": "success",
            "progress": 100
        })
    
    def on_failure(self, exc, task_id, args, kwargs, einfo):
        # 任务失败
        notify_user_via_websocket(user_id, {
            "task_id": task_id,
            "status": "failed",
            "error": str(exc)
        })

@app.task(base=DocumentProcessTask, bind=True)
def process_document(self, document_id: int):
    # 1. 解析文档
    self.update_state(state="PROGRESS", meta={"progress": 20})
    content = parse_document(document_id)
    
    # 2. 文本切分
    self.update_state(state="PROGRESS", meta={"progress": 40})
    chunks = split_text(content)
    
    # 3. 向量化
    self.update_state(state="PROGRESS", meta={"progress": 60})
    embeddings = embed_chunks(chunks)
    
    # 4. 写入 Milvus
    self.update_state(state="PROGRESS", meta={"progress": 80})
    insert_to_milvus(document_id, chunks, embeddings)
    
    # 5. 更新数据库
    self.update_state(state="PROGRESS", meta={"progress": 100})
    update_document_status(document_id, "completed")
    
    return {"document_id": document_id, "chunk_count": len(chunks)}
```
